async function recalculateAstroData() { const dateStr = document.getElementById('natal_birthdate').value; const timeStr = document.getElementById('natal_birthtime').value || "12:00"; if (!dateStr) return; const lat = selectedGeoData ? selectedGeoData.lat : -23.5505; const lon = selectedGeoData ? selectedGeoData.lon : -46.6333; const tzOffset = (selectedGeoData && selectedGeoData.tz !== undefined) ? selectedGeoData.tz : -3; try { const url = `https://motor-astrologia.vercel.app/api?data=${dateStr}&hora=${timeStr}&fuso=${tzOffset}&lat=${lat}&lon=${lon}`; const res = await fetch(url); if (res.ok) { const dados = await res.json(); if (!dados.erro) { PLANETS_DEF.forEach(p => { const info = dados[p.name]; if (info) { const idx = SIGNS.findIndex(s => s.name === info.signo); if (idx !== -1) { selectSignOption(`${p.id}_sign`, idx, false); document.getElementById(`${p.id}_deg`).value = info.grau; document.getElementById(`${p.id}_min`).value = info.minuto; } } }); if (dados.Ascendente) { const idx = SIGNS.findIndex(s => s.name === dados.Ascendente.signo); selectSignOption('asc_sign', idx, false); document.getElementById('asc_deg').value = dados.Ascendente.grau; document.getElementById('asc_min').value = dados.Ascendente.minuto; } if (dados.MC) { const idx = SIGNS.findIndex(s => s.name === dados.MC.signo); selectSignOption('mc_sign', idx, false); document.getElementById('mc_deg').value = dados.MC.grau; document.getElementById('mc_min').value = dados.MC.minuto; } if (dados.Nodo_Norte) { const idx = SIGNS.findIndex(s => s.name === dados.Nodo_Norte.signo); selectSignOption('node_sign', idx, false); document.getElementById('node_deg').value = dados.Nodo_Norte.grau; document.getElementById('node_min').value = dados.Nodo_Norte.minuto; } if (dados.Sizigia) { const idx = SIGNS.findIndex(s => s.name === dados.Sizigia.signo); selectSignOption('syz_sign', idx, false); document.getElementById('syz_deg').value = dados.Sizigia.grau; document.getElementById('syz_min').value = dados.Sizigia.minuto; } } } } catch (err) { console.error("Erro ao consultar a API no Vercel:", err); } finally { renderMandala(); } }
